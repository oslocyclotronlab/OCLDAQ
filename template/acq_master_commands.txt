#
# ---- Electronic logbook (elog) ----
# Entries are submitted directly by master (no 'elog' binary needed); other
# programs can read the same keys through command_list::get().
# SSL is detected automatically: if the server supports it, it is always used,
# otherwise the entry is sent as plain HTTP and a warning is printed.
#
# elog_port is optional; without it 443 (SSL) is tried first, then 80.
# elog_subdir is an optional URL prefix, e.g. "elog" for https://host/elog/<logbook>/.
# elog_password may be left empty; it is then taken from $ELOG_PASSWORD.
# elog_ssl_verify = yes refuses servers with an untrusted certificate
# (default: print a warning and continue).
# elog_encoding is plain (default), ELCode or HTML.
# Every elog_attr_<Name> = <value> becomes the attribute <Name> on each entry.
